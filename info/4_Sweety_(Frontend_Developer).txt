#### Step 3: Open the Camera with JavaScript (`static/js/main.js`)

Write code to request webcam permission and show live video stream:



```javascript

let video = document.getElementById('webcam');

let stream = null;



async function startCamera() {

    try {

        stream = await navigator.mediaDevices.getUserMedia({ video: true });

        video.srcObject = stream;

        console.log("Webcam connected!");

    } catch (error) {

        alert("Camera permission denied or no camera found.");

        console.error("Camera Error:", error);

    }

}



function stopCamera() {

    if (stream) {

        stream.getTracks().forEach(track => track.stop());

        video.srcObject = null;

    }

}

```



#### Step 4: Talk to Pooja's Backend (`/predict`)

Send requests to Pooja's Flask server and handle responses:



```javascript

async function getSignPrediction(landmarkData) {

    try {

        const response = await fetch('/predict', {

            method: 'POST',

            headers: { 'Content-Type': 'application/json' },

            body: JSON.stringify({ landmarks: landmarkData })

        });



        const result = await response.json(); // { status, label, confidence }



        if (result.status === "recognized") {

            document.getElementById('signLabel').innerText = result.label;

            document.getElementById('confidenceScore').innerText = (result.confidence * 100) + "%";

            appendToTranscript(result.label);

        } else if (result.status === "uncertain") {

            showConfirmationModal(result.label); // Open pop-up window

        } else if (result.status === "no_hand") {

            document.getElementById('signLabel').innerText = "No hand detected";

        }

    } catch (err) {

        console.error("Failed to connect to backend:", err);

    }

}

```



#### Step 5: Build the Uncertainty Pop-up Modal

#### Step 6: Add Text-to-Speech (TTS)

Use the browser's built-in Web Speech API to speak the transcript:



```javascript

function speakTranscript() {

    let text = document.getElementById('transcript').value;

    if (text.trim() !== "") {

        let utterance = new SpeechSynthesisUtterance(text);

        utterance.lang = 'en-IN'; // Indian English Accent

        window.speechSynthesis.speak(utterance);

    }

}

```
